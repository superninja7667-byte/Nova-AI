 function runSearch(q){
  const holder=$('#novaSearchResults');if(!holder)return;const query=q.trim().toLowerCase();if(!query){holder.innerHTML='
Type a word or phrase to search your Nova workspace.
';return}
  const results=[];if(searchFilter==='All'||searchFilter==='Chats')chats.forEach(c=>{const all=[c.title,...(c.messages||[]).map(m=>m.text)].join(' ');if(all.toLowerCase().includes(query)){const snippet=(c.messages||[]).find(m=>(m.text||'').toLowerCase().includes(query))?.text||c.title;results.push({type:'Chat',title:c.title,snippet,id:c.id})}});
  if(searchFilter==='All'||searchFilter==='Projects')projectCatalog().forEach(p=>{if([p.name,p.description,p.instructions,...(p.files||[]).map(f=>f.content)].join(' ').toLowerCase().includes(query))results.push({type:'Project',title:p.name,snippet:p.description||'Project workspace',id:p.id})});
  if(searchFilter==='All'||searchFilter==='Artifacts')artifacts.forEach(a=>{if([a.title,a.type,a.content].join(' ').toLowerCase().includes(query))results.push({type:'Artifact',title:a.title,snippet:a.content.slice(0,180),id:a.id})});
  holder.innerHTML=results.length?results.map(r=>`
${esc(r.title)}
${esc(r.snippet)}
${r.type}
`).join(''):'
No results found
Try a different phrase or choose another category.
';
 }
